# ── Why the `python_version` markers below are required ──────
# numpy, scipy, scikit-learn, matplotlib, torch and pydantic-core all
# ship compiled, per-CPython wheels, and the newest release of each is
# *not* back-ported to older interpreters. An unconditional upper cap
# such as `numpy<2.3.0` therefore excludes every version that has a
# cp313/cp314 wheel, which pushes pip into a source build that fails.
#
# Each branch below is the newest release line that actually ships a
# binary wheel for that interpreter. Lower bounds are unchanged from the
# previous revision, so nothing is upgraded more than necessary, and no
# package crosses a major-version boundary.
#
# Wheel inventories verified against PyPI; the full transitive
# resolution is re-checked in CI on every supported interpreter.

# ── Data Manipulation & Storage ──────────────────────────────
# pandas 2.3.x is the newest 2.x line and ships wheels for
# cp310–cp314, so it needs no marker and its cap is unchanged.

pyarrow>=15.0.0,<26.0.0
numpy>=1.26.0,<2.3.0; python_version < "3.11"
numpy>=1.26.0,<2.6.0; python_version >= "3.11"
scipy>=1.12.0,<1.16.0; python_version < "3.11"
scipy>=1.12.0,<1.19.0; python_version >= "3.11"
pyyaml>=6.0.1
